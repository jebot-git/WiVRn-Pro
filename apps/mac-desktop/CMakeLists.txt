cmake_minimum_required(VERSION 3.20)
project(MacDesktopRelay VERSION 0.1.0 LANGUAGES NONE)
set(CMAKE_INSTALL_LIBDIR lib CACHE PATH "Library install directory")
include(GNUInstallDirs)
find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" -c "from PySide6 import QtCore, QtGui, QtWidgets"
    RESULT_VARIABLE qt_status ERROR_VARIABLE qt_error)
if(NOT qt_status EQUAL 0)
    message(FATAL_ERROR "Mac Desktop Relay requires PySide6: ${qt_error}")
endif()
configure_file(mac-desktop-relay.in mac-desktop-relay @ONLY)
configure_file(mac-desktop-relay.desktop.in mac-desktop-relay.desktop @ONLY)
install(PROGRAMS "${CMAKE_CURRENT_BINARY_DIR}/mac-desktop-relay" DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES launcher.py receiver.py protocol.py DESTINATION ${CMAKE_INSTALL_DATADIR}/mac-desktop-relay)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/mac-desktop-relay.desktop" DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES mac-desktop-relay.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
install(FILES README.rst COPYING DESTINATION ${CMAKE_INSTALL_DOCDIR})
enable_testing()
add_test(NAME relay COMMAND "${Python3_EXECUTABLE}" -m unittest discover -s "${CMAKE_CURRENT_SOURCE_DIR}" -p "test_*.py")
set_tests_properties(relay PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
