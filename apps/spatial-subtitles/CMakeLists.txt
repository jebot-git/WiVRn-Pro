cmake_minimum_required(VERSION 3.22)
project(SpatialSubtitles VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(PkgConfig REQUIRED)
pkg_check_modules(DEPS REQUIRED IMPORTED_TARGET openxr sdl2 cairo pangocairo epoxy)
find_package(X11 REQUIRED)
find_package(Vulkan REQUIRED)
find_package(Threads REQUIRED)
add_executable(spatial-subtitles-overlay native/overlay.cpp)
target_link_libraries(spatial-subtitles-overlay PRIVATE PkgConfig::DEPS X11::X11)
target_compile_options(spatial-subtitles-overlay PRIVATE -Wall -Wextra -Wno-missing-field-initializers)
add_library(XR_APILAYER_SPATIAL_subtitles SHARED native/capture/layer.cpp native/capture/frame_ipc.cpp native/capture/vulkan_readback.cpp)
target_include_directories(XR_APILAYER_SPATIAL_subtitles PRIVATE native/capture)
target_link_libraries(XR_APILAYER_SPATIAL_subtitles PRIVATE Vulkan::Vulkan Threads::Threads)
set_target_properties(XR_APILAYER_SPATIAL_subtitles PROPERTIES CXX_VISIBILITY_PRESET hidden)
install(TARGETS spatial-subtitles-overlay RUNTIME DESTINATION bin)
install(TARGETS XR_APILAYER_SPATIAL_subtitles LIBRARY DESTINATION lib)
install(DIRECTORY python/ DESTINATION share/spatial-subtitles FILES_MATCHING PATTERN "*.py")

include(CTest)
if(BUILD_TESTING)
    add_executable(capture-layer-test tests/linux_layer_test.cpp native/capture/frame_ipc.cpp native/capture/vulkan_readback.cpp)
    target_include_directories(capture-layer-test PRIVATE native/capture)
    target_link_libraries(capture-layer-test PRIVATE XR_APILAYER_SPATIAL_subtitles Vulkan::Vulkan)
    add_test(NAME capture-layer COMMAND capture-layer-test)
    add_executable(readback-test tests/vulkan_readback_test.cpp native/capture/vulkan_readback.cpp)
    target_include_directories(readback-test PRIVATE native/capture)
    target_link_libraries(readback-test PRIVATE Vulkan::Vulkan)
    add_test(NAME readback COMMAND readback-test)
endif()
install(FILES assets/Jupiteroid-Regular.ttf assets/Jupiteroid-LICENSE.txt assets/Jupiteroid-Regular.ttf.license DESTINATION share/fonts/spatial-subtitles)
install(FILES assets/icons/spatial-subtitles.png DESTINATION share/spatial-subtitles)
