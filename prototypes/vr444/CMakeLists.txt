cmake_minimum_required(VERSION 3.20)
project(vr444 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_program(GLSLC glslc REQUIRED NO_CMAKE_FIND_ROOT_PATH)
foreach(mode encode decode)
  string(TOUPPER ${mode} MODE)
  add_custom_command(OUTPUT ${mode}.spv
    COMMAND ${GLSLC} --target-env=vulkan1.1 -O -D${MODE}=1 ${CMAKE_CURRENT_SOURCE_DIR}/codec.comp -o ${mode}.spv
    DEPENDS codec.comp VERBATIM)
endforeach()
set(SHADERS encode.spv decode.spv)
foreach(wg 64 128)
  foreach(mode encode decode)
    string(TOUPPER ${mode} MODE)
    add_custom_command(OUTPUT ${mode}-coop${wg}.spv
      COMMAND ${GLSLC} --target-env=vulkan1.1 -O -D${MODE}=1 -DWORKGROUP=${wg} ${CMAKE_CURRENT_SOURCE_DIR}/codec_coop.comp -o ${mode}-coop${wg}.spv
      DEPENDS codec_coop.comp VERBATIM)
    list(APPEND SHADERS ${mode}-coop${wg}.spv)
  endforeach()
endforeach()
add_custom_target(shaders ALL DEPENDS ${SHADERS})
add_executable(vr444 main.cpp codec.cpp foveation.cpp vulkan_runner.cpp)
add_executable(vr444-probe vulkan_probe.cpp)
target_compile_options(vr444 PRIVATE -Wall -Wextra -Wpedantic -Wno-missing-field-initializers)
if(ANDROID)
  target_link_libraries(vr444 PRIVATE vulkan)
  target_link_libraries(vr444-probe PRIVATE vulkan)
else()
  find_package(Vulkan REQUIRED)
  target_link_libraries(vr444 PRIVATE Vulkan::Vulkan)
  target_link_libraries(vr444-probe PRIVATE Vulkan::Vulkan)
endif()
add_dependencies(vr444 shaders)
enable_testing()
add_test(NAME reference COMMAND vr444 --self-test)
add_test(NAME vulkan COMMAND vr444 --gpu-test ${CMAKE_CURRENT_BINARY_DIR})
add_test(NAME foveation COMMAND vr444 --foveation-test)
foreach(wg 64 128)
  add_test(NAME vulkan_coop${wg} COMMAND vr444 --gpu-test ${CMAKE_CURRENT_BINARY_DIR})
  set_tests_properties(vulkan_coop${wg} PROPERTIES ENVIRONMENT "VR444_KERNEL=coop${wg}")
endforeach()
